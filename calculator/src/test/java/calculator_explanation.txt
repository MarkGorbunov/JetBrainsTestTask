    private static boolean isPrime(Integer candidate) {
        boolean flag = false;
        for (int i = 2; i <= candidate / 2; ++i) {
            // condition for nonprime number
            if (candidate % i == 0) {
                flag = true;
                break;
            }
        }
        return flag;
    }
